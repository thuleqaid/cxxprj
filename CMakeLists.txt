cmake_minimum_required(VERSION 4.0)

set(PROG_NAME MyProject)
set(PROG_DESC "A brief description")
project(${PROG_NAME} DESCRIPTION ${PROG_DESC} LANGUAGES C CXX)
# 使用"CMake: Select a Kit"来选择32位/64位 MinGW
# 使用"CMake: Select Variant"来选择Debug/Release
# set(CMAKE_BUILD_TYPE Release)

option(TARGET_TYPE_EXECUTABLE "Main target type, executable/library" ON)
option(DETECT_ARCH "Automatically detect 32bit/64bit" ON)
option(AUTOMATIC_SEARCH_MOD "Automatically search for modules in the module directory" ON)
option(MinGW_STATIC_LIB "Build with static gcc, stdc++ and winpthread." OFF)
option(COPY_MinGW_DLL "Copy MinGW DLLs(gcc, stdc++ and winpthread) to the build." ON)

option(DOCTEST "Test with DocTest library." ON)
option(GMOCK "Test with Google mock library." OFF)
option(GTEST "Test with Google test library." OFF)
option(UCHARDET "Enable uchardet library." OFF)
option(IMGUI "Enable ImGui library." OFF)
option(IMGUI_TE "Enable ImGui Test Engine library." OFF)
option(FREETYPE "Enable FreeType2 library." OFF)
option(ZLIB "Enable ZLib library." OFF)
option(PNG "Enable PNG library." OFF)
option(NNG "Enable NNG library." OFF)

if (NOT MinGW_STATIC_LIB)
  set(COPY_MinGW_DLL ON CACHE BOOL "Copy MinGW DLLs(gcc, stdc++ and winpthread) to the build." FORCE)
endif ()
if (GMOCK AND GTEST)
  set(GTEST OFF CACHE BOOL "Test with Google test library." FORCE)
endif ()
if (IMGUI_TE)
  set(IMGUI ON CACHE BOOL "Enable ImGui library." FORCE)
endif ()

set(CMAKE_CXX_FLAGS " -std=c++17 -Wall")
if (CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  #检查GNU是否支持-Wa,-mbig-obj
  include(CheckCXXCompilerFlag)
  check_cxx_compiler_flag("-Wa,-mbig-obj" GNU_BIG_OBJ_FLAG_ENABLE)
  message(STATUS GNU_BIG_OBJ_FLAG_ENABLE=${GNU_BIG_OBJ_FLAG_ENABLE})
  #编译选项中注入GCC版本
  set(_compiler ${CMAKE_CXX_COMPILER})
  if(NOT _compiler)
      set(_compiler ${CMAKE_C_COMPILER})
  endif()
  execute_process(
      COMMAND ${_compiler} --version
      OUTPUT_VARIABLE _FULL_OUT
      OUTPUT_STRIP_TRAILING_WHITESPACE
      RESULT_VARIABLE CMD_RESULT
  )
  set(GCC_VERSION_LINE "(unknown)")
  if(CMD_RESULT EQUAL 0 AND _FULL_OUT)
      string(REGEX MATCH "^[^\r\n]*" GCC_VERSION_LINE "${_FULL_OUT}")
  endif()
  add_definitions(-DGCC_VERSION_LINE="${GCC_VERSION_LINE}")
  message(STATUS GCC_VERSION_LINE="${GCC_VERSION_LINE}")
endif()
add_definitions(-DPROG_NAME="${PROG_NAME}")
add_definitions(-DPROG_DESCRIPTION="${PROG_DESC}")

set(ROOT_DIR ${CMAKE_CURRENT_SOURCE_DIR})
set(MODULE_DIR "${ROOT_DIR}/modules")
set(EXTLIBS)
set(EXT_LIBS_ROOT "${ROOT_DIR}/extlibs")
set(CMAKE_PREFIX_PATH ${CMAKE_PREFIX_PATH} "${EXT_LIBS_ROOT}")
include_directories(${ROOT_DIR}/include
                    ${EXT_LIBS_ROOT}/include)
if(DETECT_ARCH)
  include(CheckTypeSize)
  check_type_size("void*" MY_PTR_SIZE)
  if(MY_PTR_SIZE EQUAL 8)
    set(EXT_LIBS_ARCH_ROOT "${EXT_LIBS_ROOT}/x64")
  else()
    set(EXT_LIBS_ARCH_ROOT "${EXT_LIBS_ROOT}/x86")
  endif()
  set(CMAKE_PREFIX_PATH ${CMAKE_PREFIX_PATH} "${EXT_LIBS_ARCH_ROOT}")
  include_directories(${EXT_LIBS_ARCH_ROOT}/include)
else()
  set(MY_PTR_SIZE 0)
endif()
if(EXISTS "${ROOT_DIR}/extsrc/extlibs.cmake")
  include("${ROOT_DIR}/extsrc/extlibs.cmake")
endif()

if(IMGUI)
  set(EXTLIBS ${EXTLIBS} d3d11 d3dcompiler dxgi dwmapi)
  set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -mwindows")
  add_definitions(-DAPP_SUBSYSTEM_GUI)
  add_definitions(-DIMGUI_APP_WIN32_DX11)
  add_definitions(-DIMGUI_USER_CONFIG="${MODULE_DIR}/imgui/inc/imgui_app_imconfig.h")
  add_definitions(-DUNICODE)
  if(IMGUI_TE)
    add_definitions(-DIMGUI_MAIN_ENABLE_TEST_ENGINE)
  endif()
  if(FREETYPE)
    add_definitions(-DIMGUI_ENABLE_FREETYPE)
  endif()
endif()

set(MODULE_LIST)
if(AUTOMATIC_SEARCH_MOD)
  # 通过指定EXCLUDE_MODULE_LIST来排除需要编译的模块
  set(EXCLUDE_MODULE_LIST _tmpl_)
  if(NOT IMGUI)
    set(EXCLUDE_MODULE_LIST ${EXCLUDE_MODULE_LIST} imgui)
  endif()
  # Generate modules list
  file(GLOB dirs "${MODULE_DIR}/*")
  foreach (item ${dirs})
    if(IS_DIRECTORY ${item})
      string(CONCAT makefile ${item} "/CMakeLists.txt")
      if((EXISTS ${makefile}) AND (NOT (IS_DIRECTORY ${makefile})))
        cmake_path(GET item FILENAME moditem)
        list(FIND EXCLUDE_MODULE_LIST ${moditem} list_index)
        if(list_index LESS 0)
          set(MODULE_LIST ${MODULE_LIST} ${moditem})
        endif ()
      endif ()
    endif ()
  endforeach ()
else()
  # 直接指定需要编译的模块
  if(IMGUI)
    set(MODULE_LIST ${MODULE_LIST} imgui)
  endif()
endif()
message(STATUS "Comiple Modules=${MODULE_LIST}")

foreach (moditem ${MODULE_LIST})
  add_subdirectory(${MODULE_DIR}/${moditem})
endforeach ()

file(GLOB sources CONFIGURE_DEPENDS ${MODULE_DIR}/entry/*.cpp ${MODULE_DIR}/entry/*.cc ${MODULE_DIR}/entry/*.h)
if(TARGET_TYPE_EXECUTABLE)
  add_executable(${PROG_NAME} ${sources})
  target_compile_options(${PROG_NAME}
      PRIVATE
      $<$<CXX_COMPILER_ID:MSVC>:/bigobj>
      $<$<AND:$<CXX_COMPILER_ID:GNU>,$<BOOL:${GNU_BIG_OBJ_FLAG_ENABLE}>>:-Wa,-mbig-obj>
      )
  if(MinGW_STATIC_LIB)
    target_link_libraries(${PROG_NAME} -static gcc stdc++ winpthread -dynamic)
  endif()
else()
  add_library(${PROG_NAME} SHARED ${sources})
  target_compile_options(${PROG_NAME}
      PRIVATE
      $<$<CXX_COMPILER_ID:MSVC>:/bigobj>
      $<$<AND:$<CXX_COMPILER_ID:GNU>,$<BOOL:${GNU_BIG_OBJ_FLAG_ENABLE}>>:-Wa,-mbig-obj>
      )
  set_target_properties(${PROG_NAME} PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY_DEBUG "${ROOT_DIR}/lib/Debug"
    RUNTIME_OUTPUT_DIRECTORY_RELEASE "${ROOT_DIR}/lib/Release"
  )
  if(MinGW_STATIC_LIB)
    set(CMAKE_SHARED_LINKER_FLAGS " -static-libgcc -static-libstdc++ -static")
  endif()
endif()
target_link_options(${PROG_NAME} PRIVATE -Wl,--no-insert-timestamp)
if (CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  target_link_libraries(${PROG_NAME} -Wl,--start-group ${MODULE_LIST} ${EXTLIBS} -Wl,--end-group)
else()
  target_link_libraries(${PROG_NAME} ${MODULE_LIST} ${EXTLIBS})
endif()

if(MINGW AND COPY_MinGW_DLL)
  string(REGEX REPLACE "^(.+/)[^/]+$" "\\1" CXX_DIR ${CMAKE_CXX_COMPILER})
  if((MY_PTR_SIZE EQUAL 8) OR (MY_PTR_SIZE EQUAL 0))
    file(COPY_FILE "${CXX_DIR}/libgcc_s_seh-1.dll" "${CMAKE_BINARY_DIR}/libgcc_s_seh-1.dll" RESULT STATUS)
  endif()
  if((MY_PTR_SIZE EQUAL 4) OR (MY_PTR_SIZE EQUAL 0))
    file(COPY_FILE "${CXX_DIR}/libgcc_s_dw2-1.dll" "${CMAKE_BINARY_DIR}/libgcc_s_dw2-1.dll" RESULT STATUS)
  endif()
  file(COPY_FILE "${CXX_DIR}/libstdc++-6.dll" "${CMAKE_BINARY_DIR}/libstdc++-6.dll" RESULT STATUS)
  file(COPY_FILE "${CXX_DIR}/libwinpthread-1.dll" "${CMAKE_BINARY_DIR}/libwinpthread-1.dll" RESULT STATUS)
endif()

################################
# GTest
################################
if (GMOCK OR GTEST)
  set(TEST_ROOT_DIR "${ROOT_DIR}/tests")
  enable_testing()
  foreach (moditem ${MODULE_LIST})
    if (IS_DIRECTORY ${TEST_ROOT_DIR}/${moditem}/src)
      set(srcfiles )
      aux_source_directory(${TEST_ROOT_DIR}/${moditem}/src srcfiles)
      add_executable(test_${moditem} ${srcfiles})
      target_include_directories(test_${moditem} PRIVATE ${TEST_ROOT_DIR}/${moditem}/inc ${MODULE_DIR}/${moditem}/inc)
      target_link_libraries(test_${moditem} ${MODULE_LIST})
      if (MSVC)
        set_target_properties( test_${moditem} PROPERTIES COMPILE_OPTIONS "${COMPILE_OPTIONS} /MTd"
                                                          LINK_FLAGS "/NODEFAULTLIB:\"LIBCMTD.lib\"" )
      endif ()
      if (EXTLIBS)
        target_link_libraries(${PROG_NAME} ${EXTLIBS})
      endif ()
      if (GMOCK)
        target_link_libraries(test_${moditem} gmock_main gmock gtest)
      endif ()
      if (GTEST)
        target_link_libraries(test_${moditem} gtest_main gtest)
      endif ()
      add_test(NAME Test_${moditem} COMMAND test_${moditem})
    endif()
  endforeach ()
endif()
